function doGet(e) { if (!e) { e = { parameter: {} }; } if (!e.parameter) { e.parameter = {}; } var callback = e.parameter.callback; var action = e.parameter.action; try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Sheet1"); if (!sheet) { throw new Error("Sheet 'Sheet1' not found!"); } if (action === "getTasks") { var data = sheet.getDataRange().getValues(); var headers = data[0]; var tasks = []; for (var i = 1; i < data.length; i++) { var row = data[i]; if (row[0] !== "" && row[0] !== null && row[0] !== undefined) { var task = {}; for (var j = 0; j < headers.length; j++) { var header = headers[j]; var value = row[j]; if (value instanceof Date) { value = Utilities.formatDate(value, Session.getScriptTimeZone(), "dd-MMM"); } task[header] = value !== null && value !== undefined ? String(value) : ""; } tasks.push(task); } } var json = JSON.stringify({success: true, tasks: tasks}); if (callback) { return ContentService.createTextOutput(callback + "(" + json + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } } else if (action === "addTask") { var newRow = [ e.parameter.id || "", e.parameter.date || "", e.parameter.source || "", e.parameter.task || "", e.parameter.details || "", e.parameter.category || "", e.parameter.branch || "", // Branch column e.parameter.department || "", // Department column e.parameter.assignee || "", e.parameter.deadline || "", e.parameter.daysLeft || "", e.parameter.priority || "Medium", e.parameter.status || "" ]; sheet.appendRow(newRow); var json = JSON.stringify({success: true, message: "Task added"}); if (callback) { return ContentService.createTextOutput(callback + "(" + json + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } } else { var json = JSON.stringify({success: false, error: "No action specified. Use ?action=getTasks or ?action=addTask"}); if (callback) { return ContentService.createTextOutput(callback + "(" + json + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } } } catch (error) { var json = JSON.stringify({success: false, error: error.toString()}); if (callback) { return ContentService.createTextOutput(callback + "(" + json + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(json) .setMimeType(ContentService.MimeType.JSON); } } } function doPost(e) { if (!e) { e = { parameter: {} }; } if (!e.parameter) { e.parameter = {}; } return doGet(e); }
